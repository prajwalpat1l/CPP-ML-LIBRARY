cmake_minimum_required(VERSION 3.16)
project(cpp_ml_library VERSION 0.1.0 LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(ML_OPENMP "Use OpenMP when available" ON)
option(ML_NATIVE "Optimize for the build machine" OFF)
option(ML_FMA "Allow fused multiply-add (may change numerical results)" OFF)
option(ML_BUILD_EXAMPLES "Build the examples" ON)
option(ML_BUILD_BENCHMARKS "Build the benchmark executables" ON)
set(ML_OPT_LEVEL "2" CACHE STRING "Release optimization level (0, 1, 2, or 3)")
set_property(CACHE ML_OPT_LEVEL PROPERTY STRINGS 0 1 2 3)
if(NOT ML_OPT_LEVEL MATCHES "^[0-3]$")
    message(FATAL_ERROR "ML_OPT_LEVEL must be 0, 1, 2, or 3")
endif()

# Consumers can link this header-only library with add_subdirectory().
add_library(cpp_ml INTERFACE)
add_library(cpp_ml::cpp_ml ALIAS cpp_ml)
target_include_directories(cpp_ml INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(cpp_ml INTERFACE cxx_std_20)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cpp_ml INTERFACE "$<$<CONFIG:Release>:-O${ML_OPT_LEVEL}>")
    if(ML_NATIVE)
        target_compile_options(cpp_ml INTERFACE -march=native)
    endif()
    if(NOT ML_FMA)
        target_compile_options(cpp_ml INTERFACE -ffp-contract=off)
    endif()
endif()

if(ML_OPENMP)
    find_package(OpenMP QUIET COMPONENTS CXX)
    if(OpenMP_CXX_FOUND)
        target_link_libraries(cpp_ml INTERFACE OpenMP::OpenMP_CXX)
    else()
        message(WARNING "OpenMP was not found; building in serial mode. Use -DML_OPENMP=OFF to select serial mode explicitly.")
    endif()
endif()

if(ML_BUILD_EXAMPLES)
    add_executable(linear_regression examples/linear_regression.cpp)
    target_link_libraries(linear_regression PRIVATE cpp_ml::cpp_ml)
    add_executable(diabetes_mlp examples/diabetes_mlp.cpp)
    target_link_libraries(diabetes_mlp PRIVATE cpp_ml::cpp_ml)
endif()

if(ML_BUILD_BENCHMARKS)
    add_executable(bench_mlp benchmarks/bench_mlp.cpp)
    target_link_libraries(bench_mlp PRIVATE cpp_ml::cpp_ml)
    add_executable(bench_mlp_prof benchmarks/bench_mlp.cpp)
    target_link_libraries(bench_mlp_prof PRIVATE cpp_ml::cpp_ml)
    target_compile_definitions(bench_mlp_prof PRIVATE ML_PROFILE)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_kernels tests/test_kernels.cpp)
    target_link_libraries(test_kernels PRIVATE cpp_ml::cpp_ml)
    if(ML_FMA)
        target_compile_definitions(test_kernels PRIVATE ML_TEST_TOLERANT)
    endif()
    add_test(NAME kernels COMMAND test_kernels)
    if(ML_BUILD_EXAMPLES)
        add_test(NAME linear_regression_example COMMAND linear_regression)
    endif()
endif()
